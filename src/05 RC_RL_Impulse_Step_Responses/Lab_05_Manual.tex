\section{Background and Theory (Main Concepts)}
%--------------------------------------------------------------

\subsection{What You Need in the Main Body}

A first-order causal LTI system is governed by
\begin{equation}
\tau\frac{dy(t)}{dt} + y(t) = Kx(t).
\end{equation}
The time constant $\tau$ sets the speed of response:
\begin{itemize}
    \item At $t=\tau$, a rising step response reaches about 63.2\% of its final value.
    \item At $t=\tau$, a decaying exponential falls to about 36.8\% of its initial value.
\end{itemize}

\subsection{RC Circuit Model}

For the RC low-pass circuit with output across the capacitor,
\begin{equation}
RC\frac{dV_C}{dt} + V_C = x(t), \qquad \tau_{RC}=RC.
\end{equation}
For $x(t)=V_su(t)$,
\begin{equation}
V_C(t)=V_s\left(1-e^{-t/\tau_{RC}}\right).
\end{equation}

\subsection{RL Circuit Model Used in This Lab}

The RL hardware wiring is series $R_{\mathrm{ext}}$--$L$ to ground, with A0 sensing the \emph{junction node} between $R_{\mathrm{ext}}$ and $L$. That node is the inductor-side voltage.

To match the analysis objective (response across $R_{\mathrm{ext}}$), compute in MATLAB:
\begin{equation}
V_{R_{\mathrm{ext}}}(t) = V_{in}(t) - V_{node}(t),
\end{equation}
where $V_{node}(t)$ is measured by A0 and $V_{in}(t)$ is the known Arduino output level (approximately 5 V during the step window).

With inductor winding resistance $R_L$,
\begin{equation}
R_{\mathrm{eff}}=R_{\mathrm{ext}}+R_L, \qquad \tau_{RL}=\frac{L}{R_{\mathrm{eff}}}, \qquad K=\frac{R_{\mathrm{ext}}}{R_{\mathrm{eff}}}.
\end{equation}

\subsection{Impulse Approximation in Hardware}

A finite pulse approximates an impulse only when $T_p \ll \tau$. In this lab, use TA-assigned $T_p$ and verify $T_p/\tau \le 0.1$ when possible.

Full RC/RL derivations, detailed log-linearization equations, and extended mathematical notes are moved to Appendix A.

\addcontentsline{toc}{section}{Appendix A: Full Theory Derivations (Reference)}

\subsection*{A.1 RC Derivation}

Starting from KVL,
\begin{equation}
x(t)=V_R(t)+V_C(t)=RC\frac{dV_C}{dt}+V_C(t).
\end{equation}
Hence,
\begin{equation}
RC\frac{dV_C}{dt}+V_C=x(t), \qquad \tau_{RC}=RC.
\end{equation}
For step input $x(t)=V_su(t)$,
\begin{equation}
V_C(t)=V_s\left(1-e^{-t/\tau_{RC}}\right)u(t).
\end{equation}
For impulse response,
\begin{equation}
h_{RC}(t)=\frac{1}{\tau_{RC}}e^{-t/\tau_{RC}}u(t).
\end{equation}

\subsection*{A.2 RL Derivation}

With $R_{\mathrm{eff}}=R_{\mathrm{ext}}+R_L$ and current $i(t)$,
\begin{equation}
x(t)=L\frac{di}{dt}+R_{\mathrm{eff}}i(t).
\end{equation}
Output across the external resistor is
\begin{equation}
y(t)=V_{R_{\mathrm{ext}}}(t)=R_{\mathrm{ext}}i(t).
\end{equation}
Then,
\begin{equation}
\frac{L}{R_{\mathrm{eff}}}\frac{dy}{dt}+y(t)=\frac{R_{\mathrm{ext}}}{R_{\mathrm{eff}}}x(t),
\end{equation}
so
\begin{equation}
\tau_{RL}=\frac{L}{R_{\mathrm{eff}}}, \qquad K=\frac{R_{\mathrm{ext}}}{R_{\mathrm{eff}}}.
\end{equation}
For step input,
\begin{equation}
y(t)=KV_s\left(1-e^{-t/\tau_{RL}}\right)u(t).
\end{equation}

\subsection*{A.3 Log-Linearization}

For decay $y(t)=Ae^{-t/\tau}+C$,
\begin{equation}
\ln(y-C)=\ln A-\frac{t}{\tau}.
\end{equation}
For rise $y(t)=Y_\infty(1-e^{-t/\tau})$,
\begin{equation}
\ln(Y_\infty-y)=\ln Y_\infty-\frac{t}{\tau}.
\end{equation}
In both cases, slope $m=-1/\tau$, so $\hat{\tau}=-1/m$.

\begin{itemize}
    \item MATLAB hangs at \texttt{readline}: verify trigger character \texttt{s} is sent and sample counts match between Arduino and MATLAB.
